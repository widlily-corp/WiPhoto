# WiPhoto 2.5.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** довести WiPhoto до версии 2.5.0 — поднять отзывчивость на больших библиотеках, разгрузить god-controller, добавить command palette, manual keywords, white balance picker и undo/redo шорткаты.

**Architecture:** PyQt6 + Python 3.9, MVC. Core/controllers — без PyQt (кроме ImageInfo). Длительные операции в QThread, регистрируются в новом `controllers/thread_registry.py`. Photo controller декомпозируется на сервисы в `controllers/services/`. Кастомные исключения в `core/exceptions.py`.

**Tech Stack:** PyQt6, Pillow, OpenCV, NumPy, rawpy, imagehash, ExifTool (daemon).

**Спек:** [docs/superpowers/specs/2026-04-26-wiphoto-2.5.0-design.md](../specs/2026-04-26-wiphoto-2.5.0-design.md)

**Тесты:** в проекте отсутствует pytest-инфраструктура (см. CLAUDE.md). Каждая задача завершается smoke-проверкой через `python main.py` или ручной dry-run импортом модуля; формальные unit-тесты не пишем, чтобы не нарушать существующие договорённости. Где возможно — простые `python -c "..."` smoke-runs.

- [ ] **Step 1: Создать PixmapLRUCache**

`views/_pixmap_cache.py`:
```python
from collections import OrderedDict
from PyQt6.QtGui import QPixmap


class PixmapLRUCache:
    """LRU-кэш QPixmap с лимитом по суммарному размеру в байтах."""

    def __init__(self, max_bytes: int = 256 * 1024 * 1024) -> None:
        self._max_bytes = max_bytes
        self._items: "OrderedDict[str, QPixmap]" = OrderedDict()
        self._total = 0

    @staticmethod
    def _size_of(pix: QPixmap) -> int:
        try:
            n = pix.sizeInBytes()
            if n > 0:
                return int(n)
        except Exception:
            pass
        return max(1, pix.width() * pix.height() * (pix.depth() // 8 or 4))

    def get(self, key: str) -> QPixmap | None:
        pix = self._items.get(key)
        if pix is not None:
            self._items.move_to_end(key)
        return pix

    def put(self, key: str, pix: QPixmap) -> None:
        if key in self._items:
            self._total -= self._size_of(self._items[key])
            del self._items[key]
        self._items[key] = pix
        self._total += self._size_of(pix)
        while self._total > self._max_bytes and self._items:
            _, victim = self._items.popitem(last=False)
            self._total -= self._size_of(victim)

    def clear(self) -> None:
        self._items.clear()
        self._total = 0

    def __len__(self) -> int:
        return len(self._items)
```

`core/metadata_worker.py`:
```python
from __future__ import annotations
import queue
from PyQt6.QtCore import QThread, pyqtSignal
from core.metadata_reader import read_metadata


class MetadataReaderWorker(QThread):
    """Фоновый воркер для чтения EXIF через ExifTool daemon без блокировки UI."""

    metadata_ready = pyqtSignal(str, dict)
    metadata_failed = pyqtSignal(str, str)

    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._queue: "queue.Queue[str | None]" = queue.Queue()
        self._stop = False

    def request(self, path: str) -> None:
        self._queue.put(path)

    def stop(self) -> None:
        self._stop = True
        self._queue.put(None)

    def run(self) -> None:
        while not self._stop:
            try:
                path = self._queue.get(timeout=0.5)
            except queue.Empty:
                continue
            if path is None:
                break
            try:
                data = read_metadata(path)
                self.metadata_ready.emit(path, data or {})
            except Exception as e:
                self.metadata_failed.emit(path, str(e))
```

- [ ] **Step 1: Создать реестр**

`controllers/thread_registry.py`:
```python
from __future__ import annotations
import logging
from PyQt6.QtCore import QThread

log = logging.getLogger(__name__)


class ThreadRegistry:
    def __init__(self) -> None:
        self._threads: list[QThread] = []

    def register(self, thread: QThread) -> QThread:
        self._threads.append(thread)
        return thread

    def shutdown_all(self, wait_ms: int = 2000) -> None:
        for t in list(self._threads):
            try:
                if t.isRunning():
                    if hasattr(t, "stop"):
                        t.stop()
                    t.quit()
                    if not t.wait(wait_ms):
                        log.warning("Поток %s не завершился, terminate()", t)
                        t.terminate()
                        t.wait(500)
            except Exception as e:
                log.warning("Ошибка остановки потока: %s", e)
        self._threads.clear()
```

```python
@dataclass
class Command:
    id: str
    title: str
    callback: Callable[[], None]
    category: str = ""
    shortcut: str = ""


class CommandRegistry:
    def __init__(self) -> None:
        self._commands: dict[str, Command] = {}

    def register(self, cmd: Command) -> None:
        self._commands[cmd.id] = cmd

    def all(self) -> list[Command]:
        return list(self._commands.values())

    def find(self, query: str) -> list[Command]:
        q = query.strip().lower()
        if not q:
            return self.all()
        scored: list[tuple[int, Command]] = []
        for cmd in self._commands.values():
            hay = f"{cmd.title} {cmd.category}".lower()
            if q in hay:
                idx = hay.find(q)
                scored.append((idx, cmd))
        scored.sort(key=lambda x: (x[0], x[1].title))
        return [c for _, c in scored]
```

```python
from PyQt6.QtWidgets import QDialog, QVBoxLayout, QLineEdit, QListWidget, QListWidgetItem
from core.command_registry import CommandRegistry


class CommandPalette(QDialog):
    def __init__(self, registry: CommandRegistry, parent=None) -> None:
        super().__init__(parent)
        self.setWindowTitle("Команды")
        self.setModal(True)
        self.resize(520, 400)
        self._registry = registry

        layout = QVBoxLayout(self)
        self._input = QLineEdit()
        self._input.setPlaceholderText("Поиск команд…")
        self._list = QListWidget()
        layout.addWidget(self._input)
        layout.addWidget(self._list)

        self._input.textChanged.connect(self._refresh)
        self._input.returnPressed.connect(self._accept_current)
        self._list.itemActivated.connect(self._run_item)
        self._refresh("")

    def _refresh(self, text: str) -> None:
        self._list.clear()
        for cmd in self._registry.find(text):
            it = QListWidgetItem(f"{cmd.title}    {cmd.shortcut}".rstrip())
            it.setData(Qt.ItemDataRole.UserRole, cmd.id)
            self._list.addItem(it)
        if self._list.count():
            self._list.setCurrentRow(0)

    def _accept_current(self) -> None:
        it = self._list.currentItem()
        if it:
            self._run_item(it)

    def _run_item(self, item: QListWidgetItem) -> None:
        cmd_id = item.data(Qt.ItemDataRole.UserRole)
        cmd = next((c for c in self._registry.all() if c.id == cmd_id), None)
        self.accept()
        if cmd:
            cmd.callback()
```

```python
from __future__ import annotations
import subprocess
import logging
from pathlib import Path
from core.exceptions import MetadataWriteError

log = logging.getLogger(__name__)


def read_user_tags(path: str) -> list[str]:
    """Читать XMP:Subject через ExifTool (синхронно, для одиночных вызовов)."""
    try:
        from core.metadata_reader import read_metadata
        data = read_metadata(path) or {}
        raw = data.get("XMP:Subject") or data.get("Subject") or []
        if isinstance(raw, str):
            return [s.strip() for s in raw.split(",") if s.strip()]
        return list(raw)
    except Exception as e:
        log.warning("read_user_tags(%s): %s", path, e)
        return []


def write_user_tags(path: str, tags: list[str]) -> None:
    """Записать список тегов в XMP-dc:Subject. Для RAW — в .xmp sidecar."""
    p = Path(path)
    is_raw = p.suffix.lower() in {".cr2", ".cr3", ".nef", ".arw", ".raf",
                                  ".rw2", ".dng", ".orf", ".pef", ".srw"}
    target = str(p.with_suffix(p.suffix + ".xmp")) if is_raw else str(p)
    args = ["exiftool", "-overwrite_original"]
    if is_raw:
        args = ["exiftool", "-overwrite_original", "-o", target, "-tagsfromfile", str(p)]
    args += [f"-XMP-dc:Subject={t}" for t in tags] or ["-XMP-dc:Subject="]
    args.append(target)
    try:
        res = subprocess.run(args, capture_output=True, text=True, timeout=30)
        if res.returncode != 0:
            raise MetadataWriteError(res.stderr.strip() or "exiftool failed")
    except Exception as e:
        if isinstance(e, MetadataWriteError):
            raise
        raise MetadataWriteError(str(e)) from e
```

```python
class KeywordPanel(QWidget):
    tags_changed = pyqtSignal(str, list)  # path, tags

    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._path: str | None = None

        v = QVBoxLayout(self)
        self._list = QListWidget()
        h = QHBoxLayout()
        self._input = QLineEdit()
        self._input.setPlaceholderText("Новый тег…")
        btn_add = QPushButton("+")
        btn_del = QPushButton("−")
        h.addWidget(self._input)
        h.addWidget(btn_add)
        h.addWidget(btn_del)
        v.addWidget(self._list)
        v.addLayout(h)

        btn_add.clicked.connect(self._add)
        self._input.returnPressed.connect(self._add)
        btn_del.clicked.connect(self._remove_selected)

    def set_image(self, path: str | None, tags: list[str]) -> None:
        self._path = path
        self._list.clear()
        self._list.addItems(tags)
        self.setEnabled(path is not None)

    def _emit(self) -> None:
        if self._path is None:
            return
        tags = [self._list.item(i).text() for i in range(self._list.count())]
        self.tags_changed.emit(self._path, tags)

    def _add(self) -> None:
        t = self._input.text().strip()
        if not t:
            return
        self._list.addItem(t)
        self._input.clear()
        self._emit()

    def _remove_selected(self) -> None:
        for it in self._list.selectedItems():
            self._list.takeItem(self._list.row(it))
        self._emit()
```

```python
from __future__ import annotations
import numpy as np
from core.editing.base_tool import BaseTool
from core.editing.tool_temperature import TemperatureTool
from core.editing.tool_tint import TintTool


class WhiteBalanceTool(BaseTool):
    """Авто-WB по выбранному нейтральному пикселю.

    Принимает координаты пикселя (x, y) на исходном изображении,
    рассчитывает temperature/tint так, чтобы выбранный пиксель стал нейтральным,
    и применяет существующие инструменты Temperature и Tint.
    """

    name = "white_balance_picker"
    title = "Баланс белого (пипетка)"

    def compute_params(self, image: np.ndarray, x: int, y: int) -> tuple[float, float]:
        h, w = image.shape[:2]
        x = max(0, min(w - 1, int(x)))
        y = max(0, min(h - 1, int(y)))
        # Среднее по 5x5 окрестности
        x0, x1 = max(0, x - 2), min(w, x + 3)
        y0, y1 = max(0, y - 2), min(h, y + 3)
        patch = image[y0:y1, x0:x1].astype(np.float32)
        if patch.shape[2] >= 3:
            r = patch[..., 0].mean()
            g = patch[..., 1].mean()
            b = patch[..., 2].mean()
        else:
            return 0.0, 0.0
        # Простейшая модель: temperature ~ (b - r) / max, tint ~ (g - (r+b)/2) / max
        m = max(r, g, b, 1.0)
        temperature = float(np.clip((b - r) / m, -1.0, 1.0)) * 100.0
        tint = float(np.clip((g - (r + b) / 2.0) / m, -1.0, 1.0)) * 100.0
        # Знак инвертируем: если синий преобладал, нужно "теплее", т.е. отрицательный temp в нашей шкале
        return -temperature, -tint

    def apply_pick(self, image: np.ndarray, x: int, y: int) -> np.ndarray:
        temp, tint = self.compute_params(image, x, y)
        out = TemperatureTool().apply(image, value=temp)
        out = TintTool().apply(out, value=tint)
        return out
```

Если `BaseTool.apply` имеет другую сигнатуру — адаптировать.
